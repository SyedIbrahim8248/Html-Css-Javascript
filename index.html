7. CSS Selectors 

/* Element */

p{
    color:red;
}


/* Class */

.card{
    padding:20px;
}


/* ID */

#title{
    font-size:30px;
}


/* Attribute */

input[type="text"]{
    border:2px solid blue;
}

9. CSS Flexbox

    display:flex;

    justify-content:center;

    align-items:center;

    gap:20px;

    flex-wrap:wrap;

}

10. CSS Grid 

.container{

    display:grid;

    grid-template-columns:
    repeat(3,1fr);

    gap:20px;

}

        grid-template-columns:1fr;

    }

}
